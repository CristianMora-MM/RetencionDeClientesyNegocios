proc sql;
    create table WORK.PATRONES_CANCELACION as
    select 
        RAMO,
        put(FANUL, yymon.) as Mes_Anulacion,
        count(distinct POLIZA) as Total_Canceladas
    from GENTH.THUNICA
     where FANUL >= '01JAN2026'd 
      and FANUL <= '30SEP2026'd 
    group by 1, 2
    order by Total_Canceladas desc;     
quit;
