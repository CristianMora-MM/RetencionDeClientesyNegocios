proc sql;
    create table WORK.CANCELADOS_RECENTES as
    select 
        c.TOMADOR,
        c.NOMBRE,
        c.NOMCLI,
        t.POLIZA,
        t.RAMO,
        t.FANUL
    from AREACLI.CLIENTES c -- O la librería donde esté la foto del día
    inner join GENTH.THUNICA t 
        on c.TOMADOR = t.TOMADOR
    where t.FANUL >= '01ENE2026'd;
quit;
