proc sql;
    create table WORK.CAMBIO_LINEA_REAL as
    select 
        c.TOMADOR,
        c.NOMBRE,
        c.EMAIL,
        a.POLIZA,
        a.RAMO as Ramo_Anterior,
        b.RAMO as Ramo_Actual,
        b.FANUL as Fecha_Cambio,
        b.MOTANUL
    from GENTH.THUNICA a
    inner join GENTH.THUNICA b
        on a.POLIZA = b.POLIZA
    inner join AREACLI.CLIENTES c
        on a.TOMADOR = c.TOMADOR
    where a.RAMO ne b.RAMO
      and a.FANUL < b.FANUL
      and b.FANUL >= '01JAN2026'd;
quit;
