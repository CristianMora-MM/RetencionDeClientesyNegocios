PROC SQL;
    create table WORK.NO_RENOVADOS as
    select distinct
        c.TOMADOR,
        c.NOMBRE,
        t.POLIZA,
        t.RAMO,
        t.FANUL
    from AREACLI.CLIENTES c
    inner join GENTH.THUNICA t
      on c.TOMADOR = t.TOMADOR
    where t.FANUL < '01JUN2026'd
      and t.POLIZAS_REN = 0;
QUIT;
